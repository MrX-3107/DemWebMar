Cấu Hình Kết Nối Hệ Thống Cloud BaaS

CREATE TABLE IF NOT EXISTS public.users (id TEXT PRIMARY KEY, fullname TEXT, email TEXT UNIQUE, phone TEXT, password TEXT, role TEXT, status TEXT, address TEXT, "createdAt" TEXT, "lastLogin" TEXT); CREATE TABLE IF NOT EXISTS public.orders (id TEXT PRIMARY KEY, "userId" TEXT, "customerName" TEXT, "customerPhone" TEXT, "customerEmail" TEXT, province TEXT, address TEXT, product TEXT, qty INTEGER, addons TEXT, total TEXT, status TEXT, payment TEXT, note TEXT, "adminNote" TEXT, date TEXT, "createdAt" TEXT); CREATE TABLE IF NOT EXISTS public.support_tickets (id TEXT PRIMARY KEY, name TEXT, phone TEXT, email TEXT, topic TEXT, "topicLabel" TEXT, source TEXT, note TEXT, status TEXT, priority TEXT, "adminReply" TEXT, "createdAt" TEXT, "updatedAt" TEXT);
